"""Custom messages using %(value)s must render the provided value.

Built-in validators should pass the value they were given to ValidationError
via ``params`` so that a custom message can contain a ``%(value)s``
placeholder (as documented in ref/validators "Writing validators").
"""
from django.conf import settings

if not settings.configured:
    settings.configure(USE_I18N=False, USE_L10N=False)

from django.core.exceptions import ValidationError
from django.core.validators import (
    EmailValidator,
    ProhibitNullCharactersValidator,
    RegexValidator,
    URLValidator,
    int_list_validator,
)


def validator_name(validator):
    return getattr(validator, "__name__", type(validator).__name__)


def expect_message(validator, value, expected):
    name = validator_name(validator)
    try:
        validator(value)
    except ValidationError as exc:
        assert exc.messages == [expected], (
            "%s: expected %r but got %r (params=%r)"
            % (name, expected, exc.messages, getattr(exc, "params", None))
        )
    else:
        raise AssertionError("%s did not raise for %r" % (name, value))


# RegexValidator must expose the value to the custom message.
regex_validator = RegexValidator(
    r"^[0-9]+$", message="%(value)s is not a number."
)
expect_message(regex_validator, "blah", "blah is not a number.")

# EmailValidator must expose the value to the custom message.
email_validator = EmailValidator(
    message="%(value)s is not a valid email address."
)
expect_message(email_validator, "blah", "blah is not a valid email address.")

# URLValidator must expose the value to the custom message.
url_validator = URLValidator(message="%(value)s is not a valid URL.")
expect_message(url_validator, "blah", "blah is not a valid URL.")

# int_list_validator builds a RegexValidator, so it must too.
list_validator = int_list_validator(
    sep=",", message="%(value)s is not a valid list."
)
expect_message(list_validator, "blah", "blah is not a valid list.")

# ProhibitNullCharactersValidator must expose the value to the custom message.
null_validator = ProhibitNullCharactersValidator(
    message="%(value)s is not valid."
)
expect_message(null_validator, "a\x00b", "a\x00b is not valid.")

print("OK")

import sys

    """Call ``modify_sys_path`` with ``entry`` first and return resulting path."""

        sys.path[:] = [entry] + original_path

"""Reproduction for Sphinx issue: LaTeX inline code highlighting adds spaces.

The issue is that an inline ``code`` role using syntax highlighting produces
LaTeX where the highlighted code is surrounded by newlines inside
``\sphinxcode{\sphinxupquote{...}}``.  In LaTeX those newlines are rendered as
visible spaces, so the generated output has an unwanted space at the start and
end of the inline code.

This test checks the semantic requirement: the content directly inside the
inline-code macro must not start with whitespace, and if it ends with
whitespace, that whitespace must be suppressed by a trailing ``%`` (the usual
LaTeX way to gobble the newline).
"""

import io
import pathlib
import re
import tempfile

from sphinx.application import Sphinx


ISSUE_SOURCE = """\
.. role:: python(code)
   :language: python
   :class: highlight

Inline :python:`def foo(1 + 2 + None + "abc"): pass` code block

.. code-block:: python

   def foo(1 + 2 + None + "abc"): pass
"""


def build_latex(tmpdir: str) -> str:
    root = pathlib.Path(tmpdir)
    src = root / "src"
    out = root / "out"
    doctrees = root / "doctrees"
    src.mkdir()
    out.mkdir()
    doctrees.mkdir()

    (src / "conf.py").write_text("project = 'repro'\n", encoding="utf-8")
    (src / "index.rst").write_text(ISSUE_SOURCE, encoding="utf-8")

    status = io.StringIO()
    warning = io.StringIO()
    app = Sphinx(
        srcdir=str(src),
        confdir=str(src),
        outdir=str(out),
        doctreedir=str(doctrees),
        buildername="latex",
        status=status,
        warning=warning,
        freshenv=True,
    )
    app.build()

    tex_files = list(out.glob("*.tex"))
    assert len(tex_files) == 1, f"expected exactly one .tex file, got {tex_files!r}"
    return tex_files[0].read_text(encoding="utf-8")


def main() -> None:
    with tempfile.TemporaryDirectory() as tmpdir:
        tex = build_latex(tmpdir)

    match = re.search(
        r"\\sphinxcode\{\\sphinxupquote\{(?P<body>.*?)\}\} code block",
        tex,
        flags=re.DOTALL,
    )
    assert match is not None, (
        "could not find the highlighted inline-code macro in LaTeX output:\n"
        + tex
    )

    body = match.group("body")
    assert body, "highlighted inline code unexpectedly has no content"

    assert not body[0].isspace(), (

        f"an unwanted leading space.  Body starts with {body[:40]!r}"

        assert stripped.endswith("%"), (
            "highlighted inline code ends with unsuppressed whitespace; LaTeX "
            "will render an unwanted trailing space.  "
            f"Body ends with {body[-40:]!r}"
        )


if __name__ == "__main__":
    main()

    src = root / "src"
    out = root / "out"
    doctrees = root / "doctrees"

    out.mkdir()
    doctrees.mkdir()

        tex = build_latex(tmpdir)

    # No whitespace may appear immediately after the opening macro brace.
    bad_start = re.search(r"\\sphinxcode\{\\sphinxupquote\{[ \t\r\n]", tex)
    assert bad_start is None, (
        "highlighted inline code has whitespace immediately after the opening "
        "brace; LaTeX will render a leading space:\n"

        "before the closing braces; LaTeX will render a trailing space:\n"
        + tex[bad_end.start() - 100:bad_end.start() + 60]

    main()
